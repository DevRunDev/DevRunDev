# ERD
- dbdiagram.io

```
Table User {
  id            int       [primary key, increment]
  email         varchar(255) [unique]
  username      varchar(150)
  role          enum("Student", "Instructor", "Manager") [default: "Student"]
  is_verified   boolean   [default: true]
}

Table InstructorApplication {
  id            int       [primary key, increment]
  user_id       int       [ref: > User.id]
  qualifications text
  experience    text
  sample_video_url varchar(255)
  status        enum("PENDING", "APPROVED", "REJECTED") [default: "PENDING"]
  created_at    datetime
  updated_at    datetime
}

Table Course {
  id            int       [primary key, increment]
  instructor_id int       [ref: > User.id]
  title         varchar(255)
  description   text
  price         int       [default: 0]
  status        enum("review", "approved", "not_approved") [default: "review"]
  avg_rating    float     [default: 0.0]
  thumbnail     varchar(255) [default: "default.jpg"]
  created_at    datetime
  updated_at    datetime
}

Table Section {
  id         int       [primary key, increment]
  course_id  int       [ref: > Course.id]
  title      varchar(255)
  order      int
}

Table Lesson {
  id         int       [primary key, increment]
  section_id int       [ref: > Section.id]
  title      varchar(255)
  video_url  varchar(500)
  order      int
}

Table Enrollment {
  id          int       [primary key, increment]
  student_id  int       [ref: > User.id]
  course_id   int       [ref: > Course.id]
  enrolled_at datetime
  last_updated_at datetime
  progress    decimal(5,2) [default: 0.00]
  status      enum("in_progress", "completed", "dropped") [default: "in_progress"]
  last_watched_lesson_id int [ref: > Lesson.id]
}

Table LessonProgress {
  id             int       [primary key, increment]
  student_id     int       [ref: > User.id]
  lesson_id      int       [ref: > Lesson.id]
  completed      boolean   [default: false]
  completed_at   datetime
  last_watched_at datetime
}

Table Certificate {
  id             int       [primary key, increment]
  enrollment_id  int       [ref: > Enrollment.id]
  issued_at      datetime
  certificate_id varchar(50) [unique]
}

Table CartItem {
  id          int       [primary key, increment]
  user_id     int       [ref: > User.id]
  course_id   int       [ref: > Course.id]
  added_at    datetime
}

Table Quiz {
  id            int       [primary key, increment]
  course_id     int       [ref: > Course.id]
  section_id    int       [ref: > Section.id]
  lesson_id     int       [ref: > Lesson.id]
  instructor_id int       [ref: > User.id]
  title         varchar(200)
  description   text
  created_at    datetime
  updated_at    datetime
}

Table Question {
  id         int       [primary key, increment]
  quiz_id    int       [ref: > Quiz.id]
  text       text
  order      int
}

Table Choice {
  id         int       [primary key, increment]
  question_id int      [ref: > Question.id]
  text       varchar(200)
  is_correct boolean   [default: false]
}

Table QuizAttempt {
  id             int       [primary key, increment]
  quiz_id        int       [ref: > Quiz.id]
  student_id     int       [ref: > User.id]
  started_at     datetime
  completed_at   datetime
  score          int       [default: 0]
  total_questions int      [default: 0]
  correct_answers int      [default: 0]
  is_completed   boolean   [default: false]
}

Table Answer {
  id             int       [primary key, increment]
  attempt_id     int       [ref: > QuizAttempt.id]
  question_id    int       [ref: > Question.id]
  selected_choice_id int   [ref: > Choice.id]
  is_correct     boolean   [default: false]
}

Table Review {
  id         int       [primary key, increment]
  course_id  int       [ref: > Course.id]
  user_id    int       [ref: > User.id]
  rating     int
  comment    text
  created_at datetime
  updated_at datetime
}
```
